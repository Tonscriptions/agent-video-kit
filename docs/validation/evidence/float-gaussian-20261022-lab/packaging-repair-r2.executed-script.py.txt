def save(path,value):
    assert not path.exists(),path;path.parent.mkdir(parents=True,exist_ok=True)
    path.write_text(json.dumps(value,ensure_ascii=False,indent=2)+'\n',encoding='utf8',newline='\n')


def common(source):
    for old,new in [('native-r2','native-r3'),('media-installed-r1','media-installed-r2'),
        ('installed-r1','installed-r2'),('media-r1','media-r2'),('static-review-r1','static-review-r2')]:
        source=source.replace(old,new)
    return source


def write_code(path,source):
    ast.parse(source);path.write_text(source,encoding='utf8',newline='\n')

        for name in archive.namelist():

    assert source.keys()-shipped.keys()==source.keys()-installed_files.keys()=={'float_gaussian_NOTICE.txt'}
    assert shipped==installed_files=={n:v for n,v in source.items() if n!='float_gaussian_NOTICE.txt'}
    save(backup/'diagnostic.json',{'status':'confirmed-single-NOTICE-packaging-omission','source_resources':193,

                 'release_float_gaussian_after_complete','finish_float_gaussian_chain','float_gaussian_release_docs'):

        if stem=='finish_float_gaussian_chain':
