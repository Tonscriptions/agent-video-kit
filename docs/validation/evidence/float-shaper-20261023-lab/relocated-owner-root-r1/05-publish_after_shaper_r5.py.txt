def read(path):
    return json.loads(path.read_bytes())


def sha(path):
    return 'sha256:' + hashlib.sha256(path.read_bytes()).hexdigest()


def save(path, data):
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + '.' + str(os.getpid()) + '.tmp')
    temporary.write_text(json.dumps(data, ensure_ascii=False, indent=2)+'\n', encoding='utf8', newline='\n')
    os.replace(temporary, path)


def git(root, *args):
    command = ['git','-c','gc.auto=0','-c','maintenance.auto=false','-c','core.fsmonitor=false',*args]
    result = subprocess.run(command, cwd=root, capture_output=True,
        timeout=3600 if args and args[0]=='push' else 600, creationflags=subprocess.CREATE_NO_WINDOW)
    assert result.returncode == 0, (args, result.stderr.decode('utf8','replace')[-1500:])
    return result.stdout


def load(path, name):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def inputs():
    record = read(ROOT / EVIDENCE / 'prepare-r1.json')
    assert record['status'] == 'passed-unpublished-three-skill-documentation'
    assert len(record['prepared_files']) == 4 and len(record['validations']) == 3
    assert record['checked_local_links'] == 316 and all(v['returncode']==0 for v in record['validations'])
    rows = dict(record['prepared_files'])
    rows.update({p.relative_to(ROOT).as_posix():sha(p) for p in sorted((ROOT/EVIDENCE).rglob('*')) if p.is_file()})
    return record, rows


def verify(record, rows):
    assert all(sha(ROOT/name)==digest for name,digest in rows.items())
    assert all(sha(MAIN/name)==digest for name,digest in record['original_skill_inputs'].items())


def check_links(rows, stage):
    count = 0
    for name in rows:
        if not name.endswith('.md'): continue
        text = (stage/name).read_text(encoding='utf-8-sig')
        for raw in re.findall(r'!?\[[^\]]*\]\(([^)]+)\)', text):
            if re.match(r'^[A-Za-z][A-Za-z0-9+.-]*:',raw) or raw.startswith('#'): continue
            raw = urllib.parse.unquote(raw.split('#',1)[0])
            relative = Path(os.path.normpath(str(Path(name).parent/raw)))
            assert any((base/relative).is_file() for base in (stage,MAIN)), (name,raw)
            count += 1
    return count


def preflight():
    record, rows = inputs()
    verify(record, rows)
    expected = read(PARENT_OWNER)
    assert expected['status'] in ('waiting-for-actual-full2930-installed118', 'published-V112-full2930-installed118')
    assert len(expected['frozen_inputs']) == 331
    return record, rows
