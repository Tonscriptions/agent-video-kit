def read(path):
    return json.loads(path.read_bytes())


def write(path,text):
    path.parent.mkdir(parents=True,exist_ok=True)
    path.write_text(text.rstrip()+'\n',encoding='utf8',newline='\n')

def relink(text,name):
    prefix='../'*(len(Path(name).parts)-1)
    return re.sub(r'(!?\[[^\]]*\]\()docs/',lambda m:m[1]+prefix+'docs/',text)

def contract():
    return '''# 线性浮点曝光接入契约 v0.1

作者工程显式选择`settings.pixel_pipeline.engine=float32-native-v1`；`working_signal.transfer`须为`linear`。曝光为片段`effects`数组中的静态项：

```json
{"type":"exposure","parameters":{"amount":1,"black":0.003}}
```

`amount`/别名`exposure`范围−3..3 EV；`black`范围−1..1。值须有限，布尔、未知参数、动态通道、冲突别名、参数重复置于顶层和parameters均拒绝。遗留平铺参数可读；禁用效果不执行。每片段最多128项静态效果，不支持的启用效果返回`FLOAT_TIMELINE_UNSUPPORTED`。

参数先转换binary32；`white=Float32(2^(-amount))`，`d=abs(Float32(white-black))`，`gain=Float32(1/d)`。当d精确为零时gain=1024。每个RGB先执行Float32减黑位，再执行Float32乘增益；黑位大于白位仍使用正增益。alpha在曝光核中逐位保留；RGB可为有限负数或大于1，不做裁剪或整数编码。作者输出范围策略在后续共同交付阶段执行。Float32白位与原生exp2f在不同运行库上可能有舍入差异，必须保留原生交叉对照及具体误差边界，不能宣称任意平台逐位相同。

曝光位于素材解码/工作信号转换之后，局部蒙版和仿射之前；嵌套片段在子序列合成之后应用。效果链按作者顺序执行；正式render、区间预览与单帧预览共享FloatTimeline，持久缓存按作者/素材/运行时内容SHA失效。

输入为连续H×W×4 Float32，RGB须有限，alpha须在0..1。核上限16,777,216像素、单边16,384、单块262,144像素，默认65,536；默认及最大估算工作区768MiB。估算包含输入/输出/中间块预算，既非全进程RSS测量，也不承诺共同宿主可运行所有该尺寸。取消和错误不改写调用者输入；已分配的部分输出由本次调用持有。共同宿主、源解码和最终编码的其他预算继续适用。

本实现仅静态曝光。没有迁移默认后端，没有接通全部效果/转场/字幕/裁剪/羽化或HDR；没有据合成图案、测试通过数、标签或观看副本接受自然素材质量、剪映风格一致性、实时预览或完整产品等价。MP4观看副本继承已披露的去alpha原生转换精度边界，Float32母参考和PNG16独立验收。
'''


def documents(target,full=None,published=False):
    names=list(LATEST_DOCS)+[REPORT,CONTRACT]
    block=section(full,published)
    for name in LATEST_DOCS:
        text=(MAIN/name).read_text('utf8')
        text=re.sub(r'\n?<!-- '+MARKER+r' -->.*?<!-- /'+MARKER+r' -->\n?','\n',text,flags=re.S)
        pos=text.find('\n')+1
        write(target/name,text[:pos]+'\n'+relink(block,name)+'\n'+text[pos:])
    write(target/REPORT,report(full,published));write(target/CONTRACT,contract())
    return names


def links(target,names):
    found=[]
    for name in names:
        for value in re.findall(r'!?\[[^\]]*\]\(([^\s)]+)(?:\s+[^)]*)?\)',(target/name).read_text('utf8')):
            path=unquote(value.strip('<>').split('#')[0])
            if not path or re.match(r'^[a-zA-Z][a-zA-Z0-9+.-]*:',path):continue
            assert any((base/name).parent.joinpath(path).resolve().is_file() for base in [target,ROOT,MAIN,PARENT]),(name,path)
            found.append([name,path])
    return found


def main():
    parser=argparse.ArgumentParser();parser.add_argument('--revision',required=True);args=parser.parse_args()
    assert re.fullmatch(r'r\d+',args.revision)
    target=ROOT/'product/.runtime/float-exposure-20261022'/('readme-preview-'+args.revision)
    assert not target.exists();target.mkdir(parents=True)
    names=documents(target);checked=links(target,names)
    write(ROOT/REPORT,report());write(ROOT/CONTRACT,contract())
    inputs,_,_,installed=current()
    receipt=ROOT/REL/('readme-preview-'+args.revision+'.json');assert not receipt.exists()
    write(receipt,json.dumps({'status':'passed-local-document-links','documents':{n:sha(target/n) for n in names},
        'document_count':len(names),'links':len(checked),'installed_status':installed['status'],
        'frozen_input_count':len(inputs),'main_mutated':False,'full_suite_claimed_passed':False,
        'preview_root':str(target),'generator_sha256':sha(Path(__file__))},ensure_ascii=False,indent=2))
    print(json.dumps({'documents':len(names),'links':len(checked),'installed_status':installed['status'],'full_suite_claimed_passed':False},ensure_ascii=False))


if __name__=='__main__':main()
