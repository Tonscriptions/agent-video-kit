Reuse byte-verified CC0 source media from V35. Compare composed child frames
against fixed V35 plus independent source-blend/time math, and selected PCM

and the artistic direction of wipe/slide remain separate acceptance scopes.
"""

 def independent_pcm(scenario):

 def independent_pcm_window(a,b,reverse):
  out=np.zeros((round((b-a)*SR),2),float)
  for ca,cb,sa,sid in ((0,2.5,2,'a'),(1.5,4.5,1.5,'b'),(3.5,6,1.5,'c')):
   lo,hi=max(a,ca),min(b,cb)
   if hi<=lo:continue
   begin,end=(round((b-hi)*SR),round((b-lo)*SR)) if reverse else (round((lo-a)*SR),round((hi-a)*SR))
   signal=source_pcm[sid][round((sa+lo-ca)*SR):round((sa+hi-ca)*SR)]
   if reverse:signal=signal[::-1]
   clock=b-np.arange(begin,end)/SR if reverse else a+np.arange(begin,end)/SR
   gain=np.ones(end-begin)
   if sid in ('a','b'):gain*=1-np.clip(clock-(1.5 if sid=='a' else 3.5),0,1)
   if sid in ('b','c'):gain*=np.clip(clock-(1.5 if sid=='b' else 3.5),0,1)
   out[begin:end]+=signal*gain[:,None]
  return out
