ROOT = Path(__file__).resolve().parents[1]
W, H, FPS, TPS, SECONDS, INPUT_RATE = 192, 108, 25, 90000, 590, 8000
WINDOWS = (0, 295, 586)
CLIP_SECONDS = 4
TRUTHS = {'reference': (1., 0., 0), 'positive': (1.0015, .4135, 1), 'small-negative': (.9999, .4135, 2)}


def sha(path):
    with Path(path).open('rb') as stream:
        return 'sha256:' + hashlib.file_digest(stream, 'sha256').hexdigest()


def write(path, value):
    Path(path).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n', encoding='utf8', newline='\n')


def draw_number(draw, number, x, y, scale=2):
    digits = ('abcdef', 'bc', 'abdeg', 'abcdg', 'bcfg', 'acdfg', 'acdefg', 'abc', 'abcdefg', 'abcdfg')
    rectangles = {'a': (1,0,5,1), 'b': (5,1,6,5), 'c': (5,6,6,10), 'd': (1,10,5,11),
                  'e': (0,6,1,10), 'f': (0,1,1,5), 'g': (1,5,5,6)}
    for place, character in enumerate(f'{number:05d}'):
        for segment in digits[int(character)]:
            a, b, c, d = rectangles[segment]
            left = x + place * 8 * scale
            draw.rectangle((left+a*scale, y+b*scale, left+c*scale-1, y+d*scale-1), fill=(240,245,250,255))


def picture(source, native_frame):
    rate, offset, camera = TRUTHS[source]
    event = max(0, math.floor((native_frame / FPS - offset) / rate * FPS + 1e-8))
    image = Image.new('RGBA', (W,H), ((18,90,65,255), (24,45,110,255), (90,30,95,255))[camera])
    draw = ImageDraw.Draw(image)
    draw_number(draw, native_frame, 12, 10)
    draw_number(draw, event, 12, 40)
    for row, value in ((87,native_frame), (99,event)):
        for bit in range(16):
            x = 8 + bit * 11
            draw.rectangle((x,row,x+7,row+5), fill=(255,255,255,255) if value & (1 << bit) else (0,0,0,255))
    draw.rectangle((172,10,184,65), fill=(30+camera*60,200-camera*40,80+camera*50,255))
    return image


def main():

            [sys.executable,'-E','-B','-X','utf8','-c',bootstrap,ROOT/'product/src'])

        result=subprocess.run(command,cwd=ROOT,env=env,capture_output=True,timeout=1200,creationflags=subprocess.CREATE_NO_WINDOW)

        digest='sha256:'+hashlib.sha256(result.stdout).hexdigest()

        else:(evidence/(label+'.stdout.log')).write_bytes(result.stdout)
        record['calls'].append({'label':label,'command':command,'exit_code':result.returncode,
            'expected_exit_code':expected,'cli':cli,'wall_seconds':time.monotonic()-started,'stdout_sha256':digest});save()
        assert result.returncode==expected,(label,(result.stderr+result.stdout)[-2000:].decode('utf8','replace'))
        return result.stdout
    def cli(label,*argv,expected=0):return json.loads(run(label,[*prefix,*argv],cli=True,expected=expected))
    def decode_video(label,path):
        return run(label,[exe,'-v','error','-i',path,'-map','0:v:0','-an','-fps_mode','passthrough','-f','rawvideo','-pix_fmt','rgba','pipe:1'],binary=True)

        return run(label,[exe,'-v','error','-i',path,'-map','0:a:0','-vn','-ar','48000','-ac','2','-f','f32le','pipe:1'],binary=True)

        frames=[Image.frombytes('RGBA',(W,H),raw[i:i+size]).convert('RGB') for i in range(0,len(raw),size)]
        frames[0].save(path,save_all=True,append_images=frames[1:],duration=40,loop=0,disposal=2,optimize=False)
        with Image.open(path) as image:
            duration=0
            for index in range(image.n_frames):image.seek(index);duration+=image.info.get('duration',0)
            assert duration==12000
            return {'frames':image.n_frames,'duration_ms':duration,'from_actual_encoded_video':True}
    try:
        capability=cli('capabilities','capabilities')['data']['capabilities']['media']
        check('affine-clock-command-is-discoverable',capability['multicam_clock']['protocol']=='multicam-affine-clock-v1')
        sources={}
        source_manifest={'truths':TRUTHS,'seconds':SECONDS,'fps':FPS,'width':W,'height':H,'source_audio_rate':INPUT_RATE,'sources':{}}

            source_manifest=json.loads(args.reuse_sources.read_bytes())
            check('reused-sources-match-predeclared-truth',source_manifest['seconds']==SECONDS and
                  source_manifest['truths']=={key:list(value) for key,value in TRUTHS.items()})
            for name,row in source_manifest['sources'].items():
                sources[name]=Path(row['path']);check(name+'-reused-original-SHA',sha(sources[name])==row['sha256'])
            record['reused_source_manifest_sha256']=sha(args.reuse_sources)
        else:
            rng=np.random.default_rng(172931)

            knot_times=np.arange(knots.size)/10
            for name,(rate,offset,camera) in TRUTHS.items():
                duration=SECONDS if name=='reference' else SECONDS*rate+offset+1.
                sample_count=math.ceil(duration*INPUT_RATE);wav=work/(name+'.wav')
                with wave.open(str(wav),'wb') as stream:
                    stream.setnchannels(1);stream.setsampwidth(2);stream.setframerate(INPUT_RATE)
                    for start in range(0,sample_count,INPUT_RATE):
                        source_time=np.arange(start,min(sample_count,start+INPUT_RATE),dtype=np.float64)/INPUT_RATE
                        event_time=(source_time-offset)/rate
                        gain=np.interp(event_time,knot_times,knots,left=0.,right=0.)
                        signal=gain*(.58*np.sin(2*np.pi*233*event_time)+.19*np.sin(2*np.pi*391*event_time))
                        signal[(event_time<0)|(event_time>=SECONDS)]=0.
                        stream.writeframes(np.rint(signal*32767).astype('<i2').tobytes())
                frame_count=math.ceil(duration*FPS);target=assets/('source-'+name+'.mkv')
                command=[exe,'-hide_banner','-v','error','-f','rawvideo','-pix_fmt','rgba','-s',f'{W}x{H}',
                    '-r',str(FPS),'-i','pipe:0','-i',str(wav),'-map','0:v:0','-map','1:a:0',
                    '-c:v','ffv1','-level','3','-pix_fmt','bgra','-c:a','pcm_s16le','-shortest',str(target)]
                started=time.monotonic()
                with (evidence/('make-'+name+'.stderr.log')).open('xb') as stderr:
                    process=subprocess.Popen(command,cwd=ROOT,env=env,stdin=subprocess.PIPE,stdout=subprocess.DEVNULL,
                                             stderr=stderr,creationflags=subprocess.CREATE_NO_WINDOW)
                    try:
                        for index in range(frame_count):process.stdin.write(picture(name,index).tobytes())
                    finally:process.stdin.close()
                    code=process.wait()
                record['calls'].append({'label':'make-'+name,'command':command,'exit_code':code,'cli':False,
                    'wall_seconds':time.monotonic()-started,'streamed_RGBA_frames':frame_count});save()
                assert code==0
                sources[name]=target
                packets=json.loads(run(name+'-packet-PTS',[probe,'-v','error','-select_streams','v:0','-show_packets',
                    '-show_streams','-show_entries','packet=pts:stream=time_base','-of','json',target]))
                timebase=Fraction(packets['streams'][0]['time_base'])
                check(name+'-all-original-native-PTS',[int(row['pts'])*timebase for row in packets['packets']]==
                    [Fraction(index,FPS) for index in range(frame_count)],packet_count=frame_count)
                source_manifest['sources'][name]={'path':str(target),'sha256':sha(target),'bytes':target.stat().st_size,
                    'frames':frame_count,'native_frame_interval':'1/25','PCM_samples':sample_count,'creator':'agent-video-kit',
                    'license':'CC0-1.0','origin':'Original seven-segment/binary counters and deterministic local signal; no external media.'}
        write(evidence/'source-manifest.json',source_manifest);record['source_manifest_sha256']=sha(evidence/'source-manifest.json')
        # Independent source normalization and source ranges, outside the product renderer.
        graph='[0:a]aresample=48000:first_pts=0,aformat=sample_fmts=fltp:sample_rates=48000:channel_layouts=stereo,asplit=3[x0][x1][x2];'
        graph+=';'.join(f'[x{i}]atrim=start_sample={left*48000}:end_sample={(left+CLIP_SECONDS)*48000},asetpts=PTS-STARTPTS[a{i}]' for i,left in enumerate(WINDOWS))
        graph+=';[a0][a1][a2]concat=n=3:v=0:a=1[a]'
        expected_pcm=run('independent-reference-master-PCM',[exe,'-v','error','-i',sources['reference'],
            '-filter_complex',graph,'-map','[a]','-f','f32le','pipe:1'],binary=True)
        check('independent-master-sample-count',len(expected_pcm)==12*48000*2*4)
        record['GIF']={}
        for case in ('positive','small-negative'):
            true_rate,true_offset,camera=TRUTHS[case];mode_results={}
            for mode in ('offset','affine'):
                label=case+'-'+mode;project=work/('project-'+label)
                cli(label+'-init','project','init',project)

                    cli(label+'-ingest-'+asset,'media','ingest','--project',project,'--input',path,'--asset-id',asset,
                        '--source-url','local://affine-clock/original','--license-spdx','CC0-1.0',
                        '--license-url','local://affine-clock/CC0-1.0','--verification-status','verified',

                data=json.loads((project/'project.json').read_bytes())
                data['settings']['canvas'].update(width=W,height=H,fps=FPS)

                operations=[{'op':'append_clip','track_id':'track_audio_1','id':'master_'+str(i),'asset_id':'reference',

                    plan_path=project/('switch-'+str(index)+'.json')

                    common=['media','multicam','--project',project,'--source-asset-id','reference','--source-asset-id','camera',

if __name__=='__main__':main()
