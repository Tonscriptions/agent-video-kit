"""Record actual source/media, manually viewed contact, failure and discovery."""
import argparse
import ast
from datetime import datetime,timezone
import json
from pathlib import Path
import subprocess
import sys

ROOT=Path(__file__).resolve().parents[3]
MAIN=ROOT.parents[2]
REL=ROOT/'docs/validation/evidence/float-tone-20261022-lab'
ASSETS=ROOT/'docs/assets/float-tone-20261022-r1'
sys.path.insert(0,str(ROOT/'scripts'))
from run_frozen_regression import frozen,sha


def read(path):return json.loads(path.read_bytes())


def save(path,value):
    assert not path.exists(),path
    path.write_text(json.dumps(value,ensure_ascii=False,indent=2)+'\n',encoding='utf8')


def main():
    parser=argparse.ArgumentParser();parser.add_argument('--contact-viewed',action='store_true',required=True);parser.parse_args()
    inputs=frozen();assert len(inputs)==320
    native=read(REL/'native-r2/verification.json');media=read(REL/'media-r1/verification.json')
    assert native['status']=='passed' and native['methods']==372 and native['skipped']==0
    assert native['frozen_before']==native['frozen_after']==inputs
    assert media['status']=='passed-public-CLI-floating-basic-tone-and-native-float-previews'
    assert media['frozen_before']==media['frozen_after']==inputs and all(c['ok'] for c in media['checks'])
    sys.path.insert(0,str(ROOT/'product/src'))
    from ai_ad_editor.runtime import configure_local_runtime
    configure_local_runtime()
    from PIL import Image
    with Image.open(ASSETS/'comparison.gif') as im:
        clocks=[]
        for i in range(im.n_frames):im.seek(i);clocks.append(im.info.get('duration'))
        assert im.n_frames==14 and clocks==[100]*14
    save(REL/'static-review-r1.json',{'status':'passed-static-contact-and-GIF-clock-review',
        'at_utc':datetime.now(timezone.utc).isoformat(),'contact_sha256':sha(ASSETS/'contact.png'),
        'GIF_sha256':sha(ASSETS/'comparison.gif'),'GIF_frames':14,'GIF_frame_duration_ms':clocks,
        'viewed_panels':6,'viewed_author_frames':[3,10,16],
        'view_tool':'functions.exec -> tools.view_image; exact contact viewed before this script',
        'visible_observations':['Moving rectangle, colored grid and five-bit frame markers remain present in all six panels.',
            'Author frame3 color changes and frame10 warm global-HSL rectangle preserve the source geometry.',
            'Author frame16 native-coefficient/exposure output is bright amber with magenta/yellow highlights; viewing copy uses explicit delivery clip.'],
        'continuous_playback_review':False,'listening_review':False,'natural_quality_accepted':False,
        'HDR_accepted':False,'full_capcut_parity':False})
    first=read(REL/'native-r1/verification.json')
    assert first['methods']==44 and first['failures']==0 and first['errors']==1 and first['skipped']==0
    changed=sorted(n for n,v in inputs.items() if first['frozen_after'].get(n)!=v)
    assert changed==['product/tests/test_float_tone.py']
    log=(REL/'native-r1/unittest.log').read_text('utf8')
    assert 'Bezier须提供2或4个控制值' in log
    old=(REL/'native-r1/executed-source/product/tests/test_float_tone.py').read_text('utf8')
    new=(ROOT/'product/tests/test_float_tone.py').read_text('utf8')
    assert old.replace("track('amount',0,1,mode)","track('amount',0,1,mode,controls=[1/3,2/3] if mode=='bezier' else None)")==new
    save(REL/'failure-analysis-r1.json',{'status':'passed-preserved-failure-and-scoped-correction-review',
        'first_source_sha256':sha(REL/'native-r1/verification.json'),'accepted_source_sha256':sha(REL/'native-r2/verification.json'),
        'first_source_methods':44,'first_source_seconds':first['seconds'],'first_source_failures':0,'first_source_errors':1,'first_source_skips':0,
        'source_changes_since_first_run':changed,'runtime_or_tolerance_changed':False,
        'test_cause':'The test omitted the required two/four Bezier control values; the product correctly rejected it.',
        'test_fix':'Only provide direct linear Bezier controls 1/3 and 2/3 in the six-easing fixture.',
        'native_fractional_code_limit':1.01,'media_linear_tolerance':2e-6,
        'failure_receipts_overwritten':False,'full_capcut_parity':False})
    reference=MAIN/'product/.runtime/tonal-reference-20261022-r1/verification.json'
    ref=read(reference);assert ref['status']=='preserved-incomplete-reference' and len(ref['requests'])==3
    save(REL/'source-reference-r1.json',{'status':'preserved-incomplete-local-network-source-capture',
        'original_receipt_sha256':sha(reference),'capture':ref,
        'web_reader_eq_n8_1_read':True,'local_upstream_RELEASE':'8.0.git',
        'actual_native_binary_version':'8.1','claimed_all_n8_1_sources_downloaded':False,
        'runtime_network':False,'reference_is_product_execution_proof':False})
    bootstrap=('import sys,json,unittest;sys.path.insert(0,sys.argv[1]);'
        'from ai_ad_editor.runtime import configure_local_runtime;configure_local_runtime();'
        'loader=unittest.TestLoader();suite=loader.discover(sys.argv[2]);'
        'print(json.dumps({"discovered_methods":suite.countTestCases(),"loader_errors":loader.errors,"methods_run":0}));'
        'raise SystemExit(0 if not loader.errors else 1)')
    command=[sys.executable,'-E','-B','-X','utf8','-c',bootstrap,str(ROOT/'product/src'),str(ROOT/'product/tests')]
    p=subprocess.run(command,cwd=ROOT,capture_output=True,timeout=120,creationflags=subprocess.CREATE_NO_WINDOW)
    (REL/'discovery-r1.stdout.log').write_bytes(p.stdout);(REL/'discovery-r1.stderr.log').write_bytes(p.stderr)
    assert p.returncode==0,p.stderr[-1500:]
    discovery=json.loads(p.stdout);assert discovery['discovered_methods']==2667 and discovery['methods_run']==0
    save(REL/'discovery-r1.json',{'status':'passed-discovery-only-not-suite-execution',**discovery,
        'command':command,'exit_code':p.returncode,'stdout_sha256':sha(REL/'discovery-r1.stdout.log'),
        'stderr_sha256':sha(REL/'discovery-r1.stderr.log'),'complete_suite_passed':False})
    scripts={p.relative_to(ROOT).as_posix():sha(p) for p in (ROOT/'scripts').glob('*.py')};assert len(scripts)==16
    for name in scripts:ast.parse((ROOT/name).read_text('utf8'))
    owners=('run_float_tone_complete','publish_float_tone','release_float_tone_after_complete','finish_float_tone_chain')
    text={n:(ROOT/'scripts'/(n+'.py')).read_text('utf8') for n in owners}
    assert 'published-V105-full2623-installed528' in text['run_float_tone_complete']
    assert "'media-r1/verification.json'" in text['run_float_tone_complete']
    assert 'published-V106-full2667-installed572' in text['finish_float_tone_chain']
    assert 'passed-fresh-eight-wheel-191-resources-real-previews-and-572-methods' in text['publish_float_tone']
    assert "'test_float_selective_hsl','test_float_tone']" in (ROOT/'scripts/verify_float_tone_install.py').read_text('utf8')
    save(REL/'release-driver-review-r1.json',{'status':'passed-AST-and-static-gate-review-not-execution',
        'ROOT_scripts':scripts,'script_count':len(scripts),'required_actual_parent':'published V105/full2623/installed528',
        'candidate_full_target':2667,'candidate_installed_target':572,'candidate_resource_target':191,
        'source_media_binding':'media-r1/verification.json','full_suite_executed':False,
        'force_push':False,'full_capcut_parity':False})
    assert frozen()==inputs
    (REL/'observations-r1.executed-script.py.txt').write_bytes(Path(__file__).read_bytes())
    print(json.dumps({'status':'passed-actual-source-media-and-contact-observations','source_methods':372,
        'media_checks':len(media['checks']),'GIF_frames':14,'viewed_panels':6,'discovered_methods':2667,'complete_methods_run':0}))


if __name__=='__main__':main()
