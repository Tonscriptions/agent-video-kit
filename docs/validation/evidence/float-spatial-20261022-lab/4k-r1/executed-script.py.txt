def sha(path):
    h = hashlib.sha256()
    with path.open('rb') as stream:
        while block := stream.read(1024*1024): h.update(block)
    return 'sha256:'+h.hexdigest()


def gaussian_row(value, radius):
    """Finite geometric convolutions of boundary terms; no recursive sweeps."""
    import numpy as np
    variance = radius*radius/2
    q = 4*variance/(math.sqrt(1+4*variance)+1)**2
    weights = np.power(q, np.arange(len(value), dtype=np.float64))
    forward = np.stack([(1-q)*np.convolve(value[:, c], weights)[:len(value)]
                        + q*value[0, c]*weights for c in range(4)], axis=1)
    reverse = np.stack([(1-q)*np.convolve(forward[::-1, c], weights)[:len(value)]
                        + q*forward[-1, c]*weights for c in range(4)], axis=1)
    return reverse[::-1].astype(np.float32)


def mean_row(value, radius, power, boundary, premultiplied):
    """Direct weighted taps and explicit boundary indices, not product sums."""
    import numpy as np
    data = value.astype(np.float64)
    if premultiplied: data[:, :3] *= data[:, 3:4]
    whole = math.floor(radius); fraction = radius-whole
    taps = [(offset, 1.) for offset in range(-whole, whole+1)]
    if fraction: taps += [(-whole-1, fraction), (whole+1, fraction)]
    for _ in range(power):
        result = np.zeros_like(data)
        for offset, weight in taps:
            indices = np.arange(len(data))+offset
            if boundary == 'edge': indices = np.clip(indices, 0, len(data)-1)
            else:
                indices %= 2*len(data)
                indices = np.where(indices >= len(data), 2*len(data)-1-indices, indices)
            result += data[indices]*(weight/(2*radius+1))
        data = result
    if premultiplied:
        positive = data[:, 3] > 0
        np.divide(data[:, :3], data[:, 3:4], out=data[:, :3], where=positive[:, None])
        data[:, :3][~positive] = 0
    return data.astype(np.float32)
