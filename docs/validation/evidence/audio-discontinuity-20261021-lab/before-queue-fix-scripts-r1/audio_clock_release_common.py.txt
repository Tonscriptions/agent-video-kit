def read(path):
    return json.loads(Path(path).read_bytes())

def save(path, value):
    path = Path(path); path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2)+'\n', encoding='utf8', newline='\n')

def verify(root, values):
    for name, digest in values.items():
        assert sha(Path(root)/name) == digest, name

def git(*args):
    result = subprocess.run(['git', '-c', 'gc.auto=0', '-c', 'maintenance.auto=false', *map(str,args)],
        cwd=MAIN, capture_output=True, timeout=1200, creationflags=subprocess.CREATE_NO_WINDOW)
    assert result.returncode == 0, result.stderr.decode('utf8', 'replace')[-1200:]
    return result.stdout

def links(source, text, roots):
    result = []
    for target in re.findall(r'!?\[[^\]]*\]\(([^)\n]+)\)', text):
        target = target.strip('<>').split('#', 1)[0]
        if not target or re.match(r'^[A-Za-z][\w+.-]*:', target):
            continue
        resolved = (MAIN/source).parent.joinpath(target).resolve()
        assert resolved.is_relative_to(MAIN), (source, target)
        relative = resolved.relative_to(MAIN)
        selected = next((root/relative for root in roots if (root/relative).exists()), None)
        assert selected is not None and selected.is_file(), (source, target)
        result.append({'source': source, 'target': relative.as_posix(), 'sha256': sha(selected)})
    return result

    assert full['external_fixture_manifest_unchanged'] and full['external_fixture_count'] == 8

        assert git('ls-remote', '--heads', 'origin', 'main').decode().split()[0] == PARENT_COMMIT
        assert not git('status', '--porcelain', '-z'), 'Preserve unexpected main changes'
        spec = importlib.util.spec_from_file_location('audio_clock_published_parent', MAIN/'scripts/run_frozen_regression.py')
        module = importlib.util.module_from_spec(spec); spec.loader.exec_module(module)
        assert module.frozen() == previous
    epoch = read(REL/'current-epoch-r1.json')
    assert epoch['frozen_inputs'] == len(current) and epoch['current_native_frozen'] == current
    for name, expected in (('kernel-r2.json', 48), ('paths-r2.json', 12), ('compatibility-r2.json', 147),
                            ('discovery-r1.json', 1), ('surfaces-r1.json', 152)):
        item = read(REL/name); result = item['result']
        assert item['status'] == 'passed' and item['exit_code'] == 0 and item['source_unchanged']
        assert result['terminal_success'] and result['methods'] == result['discovered_methods'] == expected
        assert result['skipped'] == result['errors'] == result['failures'] == 0
        assert item['frozen_before'] == item['frozen_after'] and sha((REL/name).with_suffix('.log')) == item['log_sha256']
        assert item['script_sha256'] == sha((REL/name).with_suffix('.script.py.txt'))

        differences = sorted(p for p in current if current[p] != item['frozen_before'][p])
        if name in epoch['related_records']:

        else:

    extra = read(REL/'extra-native-after-r1/verification.json')

    assert extra['frozen_before'] == extra['frozen_after']
    extra_epoch = sorted(p for p in current if current[p] != extra['frozen_before'][p])

    assert extra['script_sha256'] == sha(ROOT/'scripts/probe_audio_extra_native.py')
    native = read(REL/'native-r1/verification.json'); installed = read(REL/'installed-r1/verification.json')
    fresh = read(REL/'installed-r1/media/verification.json')
    for item in (native, fresh):
        assert item['status'] == 'passed' and item['frozen_before'] == item['frozen_after'] == current
        assert item['generator_sha256'] == sha(ROOT/'scripts/smoke_audio_discontinuity.py')
        assert all(row['ok'] for row in item['checks']) and not item['full_capcut_parity']
    assert native['cli_calls'] == 32 and native['other_native_commands'] == 22 and len(native['checks']) == 37
    assert installed['status'] == 'passed' and installed['frozen_before'] == installed['frozen_after'] == current

    assert installed['bundle']['wheels'] == 8 and len(installed['resource_hashes']) == 163

                and not p.relative_to(REL).as_posix().startswith(('audit-', 'full-', 'external-fixtures-', 'before-full-'))}
    return {'status': 'passed-source-and-evidence-preflight', 'frozen_inputs': current,
