def sha(path): return 'sha256:'+hashlib.sha256(Path(path).read_bytes()).hexdigest()
def read(path): return json.loads(Path(path).read_bytes())
def core():
    paths=sorted([p for folder in ('product/src','product/tests') for p in (MAIN/folder).rglob('*')
        if p.is_file() and '__pycache__' not in p.parts and p.suffix in ('.py','.mjs','.json','.ttf')])
    return {p.relative_to(MAIN).as_posix():sha(p) for p in paths}

files=[row['filename'] for row in delta['files']]
